function palette(hero, a, M) {
  const id=a.championId, skin=a.skinId
  const silver=['celestial-guard','lunar-phantom','frost-relic'].includes(skin)
  M.trim=pbrMaterial({color:silver?'#9caab7':skin==='ashen-oath'?'#89613c':'#94713b',metalness:.85,roughness:.39,envMapIntensity:.6})
  M.dark=pbrMaterial({color:'#24292c',metalness:.4,roughness:.7})
  M.edge=pbrMaterial({color:silver?'#aab4bf':'#899299',metalness:.9,roughness:.34,envMapIntensity:.55})
  M.fabric=clothMaterial(a.secondary); M.fabric.normalScale.setScalar(.09); M.fabric.map=null
  M.lining=clothMaterial(skin==='desert-oracle'?'#c6b18c':skin==='ashen-oath'?'#675645':'#bdb49d'); M.lining.normalScale.setScalar(.09); M.lining.map=null
  M.jewel=pbrMaterial({color:a.glow,metalness:.25,roughness:.19,emissive:a.glow,emissiveIntensity:.15})
  M.skin.color.set(id==='caelan'?'#a37558':id==='nyra'?'#6d412e':'#80533a')
  M.skin.roughness=.78; M.skin.envMapIntensity=.25
  M.lips=pbrMaterial({color:id==='caelan'?'#91614f':'#704037',roughness:.85})
  M.eye=pbrMaterial({color:'#9d9481',roughness:.5})
  M.iris=pbrMaterial({color:id==='caelan'?'#5c5944':'#453425',roughness:.2})
  M.hair.color.set(id==='caelan'?'#251b17':'#19171b'); M.hair.roughness=.8
  M.hairLight=pbrMaterial({color:id==='serai'?'#625950':'#302722',roughness:.84})
  M.bronze.metalness=id==='nyra'?.45:.62; M.bronze.roughness=.5; M.bronze.normalScale.setScalar(.13)
  M.leather.normalScale.setScalar(.18); M.leatherDark.normalScale.setScalar(.18)
  if(id==='orren') {
    M.stone=stoneMaterial(a.primary);M.stone.normalScale.setScalar(.28)
    M.root=woodMaterial(skin==='furnace-king'?'#3b3430':'#5c5240');M.root.normalScale.setScalar(.2)
    M.moss=pbrMaterial({color:'#586641',roughness:1})
    if(skin==='furnace-king') { M.stone.metalness=.7;M.stone.roughness=.65 }
    if(skin==='frost-relic') { M.stone.roughness=.4;M.stone.metalness=.15 }
    M.primary=M.stone;M.secondary=M.fabric
  }
  hero.secondaryMotion=[]; hero.equipment=[]
  hero.championId=id;hero.skinId=skin;hero.modelVersion=2
  hero.torsoWidth=1
}

function face(hero, M, { female=false, stone=false } = {}) {
  const h=hero.head=node(hero.hips,0,stone?.94:.92,stone?.035:0)
  h.name=stone?'Ancient carved face':'Sculpted face'
  const skin=stone?M.stone:M.skin
  const width=stone?1.36:female?.93:1
  body(h,skin,[[-.22,.062,.066],[-.1,.068,.072]])
  const geo=surface([[-.137,.052,.061,.008],[-.117,.079,.078,.012],[-.068,.100,.091],[.005,.115,.102],[.07,.111,.102,-.006],[.124,.088,.081,-.012],[.15,.015,.023,-.01]],32,0,stone?.026:0)
  // Face planes: temples, cheekbones, eye sockets, a defined chin and brow.
  const p=geo.attributes.position
  for(let i=0;i<p.count;i++) {
    let x=p.getX(i),y=p.getY(i),z=p.getZ(i)
    if(z>0) {
      const g=(cx,cy,wx,wy)=>Math.exp(-(((Math.abs(x)-cx)/wx)**2)-(((y-cy)/wy)**2))
      z += .011*g(.073,-.022,.034,.028) - .009*g(.051,.027,.029,.020) + .008*g(.047,.058,.04,.018)
      z += .016*g(0,-.12,.05,.024)
    }
    p.setXYZ(i,x*width,y*(stone?1.14:1),z*(stone?1.2:1))
  }
  geo.computeVertexNormals();part(h,geo,skin)
  for(const s of [-1,1]) {
    oval(h,skin,s*.111*width,-.01,0,.018,.037,.025,18)
    oval(h,stone?M.dark:M.lips,s*.116*width,-.008,.012,.007,.023,.008,12)
    const eye=oval(h,M.dark,s*.049*width,.027,.091,.024,.009,.012,18);eye.rotation.z=s*.08
    oval(h,stone?M.glow:M.eye,s*.049*width,.027,stone?.118:.101,.017,stone?.005:.0045,.007,18)
    if(!stone) {
      oval(h,M.iris,s*.047*width,.027,.107,.007,.005,.004,12)
      oval(h,M.dark,s*.047*width,.027,.110,.0032,.004,.002,10)
      strand(h,M.hair,[[s*.025,.049,.104],[s*.052,.058,.1],[s*.083,.051,.086]],female?.005:.007,.4)
      strand(h,skin,[[s*.026,.035,.105],[s*.049,.039,.106],[s*.074,.032,.096]],.005,.8)
    } else {
      strand(h,skin,[[s*.024,.081,.118],[s*.064,.083,.13],[s*.124,.065,.079]],.019,.4)
      strand(h,M.dark,[[s*.1,.06,.075],[s*.095,-.028,.097],[s*.126,-.085,.05]],.003,.4)
    }
  }
  body(h,skin,[[-.039,.020,.014,.104],[-.025,female?.023:.028,.024,.112],[.027,.014,.015,.099],[.065,.009,.009,.092]])
  if(!stone) {
    for(const s of [-1,1]) oval(h,M.lips,s*.015,-.035,.121,.006,.003,.003,10)
    strand(h,M.lips,[[-.034,-.078,.087],[-.012,-.073,.106],[0,-.075,.11],[.014,-.073,.102],[.035,-.078,.087]],.005,.6)
    oval(h,M.lips,0,-.084,.102,.022,.005,.006,16)
    // Fine facial hair stays close to the jaw, with no separate beard primitive.
    if(!female) for(let i=0;i<20;i++) {
      const a=-1.2+i*2.4/19
      strand(h,i%3?M.hair:M.hairLight,[[Math.sin(a)*.081,-.112+Math.abs(a)*.007,Math.cos(a)*.091+.008],[Math.sin(a)*.068,-.131+Math.abs(a)*.008,Math.cos(a)*.082+.012]],.0018,.35)
    }
  } else {
    strand(h,M.dark,[[-.052,-.098,.119],[0,-.085,.138],[.052,-.098,.119]],.005,.8)
    body(h,skin,[[-.17,.062,.064,.036],[-.137,.097,.093,.025],[-.11,.087,.083,.017]],0,0,0,.05)
  }
  return h
}

function hair(hero, M, style) {
  const head=hero.head, cap=node(head)
  cap.name=style==='braids'?'Crown braids and long plait':style==='curls'?'Coiled hair and silver temples':'Swept dark hair'
  // Hairline wraps the skull. Individual locks overlap a continuous scalp.
  body(cap,M.hair,[[.075,.108,.099,-.018],[.13,.097,.094,-.012],[.168,.063,.067,-.018],[.18,.008,.012,-.02]])
  for(const s of [-1,1]) oval(cap,M.hair,s*.103,.057,-.031,.020,.069,.067,18)
  if(style==='swept') {
    oval(cap,M.hair,-.018,.129,-.025,.108,.051,.092,24)
    for(let i=0;i<18;i++) {
      const x=(i-8.5)*.011,j=Math.sin(i*2.37)
      strand(cap,i%5?M.hair:M.hairLight,[[x,.102+j*.022,.085],[x-.033,.151+j*.012,.064],[x-.038,.173+j*.011,-.018],[x*.8,.08,-.118]],.013+(i%3)*.002,.15)
    }
    for(const s of [-1,1]) for(let i=0;i<4;i++) strand(cap,M.hair,[[s*.103,.11-i*.014,-.025],[s*.119,.042-i*.015,-.077],[s*.092,-.063-i*.013,-.079]],.013,.15)
    strand(cap,M.hair,[[.067,.162,.049],[.077,.123,.087],[.037,.061,.104]],.014,.08)
    strand(cap,M.hair,[[-.045,.164,.064],[-.085,.109,.088],[-.07,.073,.099]],.014,.06)
  } else if(style==='braids') {
    for(let i=-4;i<=4;i++) strand(cap,i%2?M.hairLight:M.hair,[[i*.023,.085,.068],[i*.023,.163,.019],[i*.017,.163,-.085],[0,.102,-.133]],.009,.7)
    const braid=node(head,.07,.101,-.117);braid.name='Articulated long braid'
    oval(braid,M.hair,0,-.018,-.019,.07,.07,.064,18)
    for(let j=0;j<3;j++) {
      const points=Array.from({length:31},(_,i)=>{
        const t=i/30,a=t*PI*11+j*PI*2/3, r=.031*(1-t*.5)
        return [Math.sin(a)*r+t*.13,-t*.69-.055,-.027-Math.sin(t*PI)*.055+Math.cos(a)*r+t*.24]
      })
      // More curve subdivisions preserve the interwoven strands at close range.
      const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)))
      part(braid,new THREE.TubeGeometry(curve,64,.018,6,false),j===1?M.hairLight:M.hair)
    }
    for(const t of [.1,.44,.67]) ring(braid,M.trim,.035,.006,t/.69*.13,-t-.055,-.027-Math.sin(t/.69*PI)*.055+t/.69*.24).rotation.x=PI/2
    hero.secondaryMotion.push({group:braid,phase:2,amount:.10,axis:'z',base:0})
  } else {
    oval(cap,M.hair,0,.103,-.084,.127,.072,.088,20)
    for(let i=0;i<26;i++) {
      const a=i*2.39996, r=.088+(.022*(i%3)), y=.10-(i%5)*.025
      const x=Math.sin(a)*r,z=-.032-Math.abs(Math.cos(a))*.087
      const points=Array.from({length:19},(_,j)=>{
        const t=j/18
        return [x+Math.sin(t*PI*6+i)*.014,y-t*(.17+(i%4)*.026),z+Math.cos(t*PI*6+i)*.014-t*.028]
      })
      const curve=new THREE.CatmullRomCurve3(points.map(p=>new THREE.Vector3(...p)))
      part(cap,new THREE.TubeGeometry(curve,24,.010,5,false),i%6===0?M.hairLight:M.hair)
    }
  }
}
